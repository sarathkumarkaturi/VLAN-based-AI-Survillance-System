! Access Switch Configuration (e.g. Student Access)
enable
configure terminal
hostname Access-Student

! Create VLANs
vlan 10
 name Student-WiFi
vlan 20
 name Faculty-Admin

! Uplink to Distribution Switch
interface GigabitEthernet0/1
 switchport trunk encapsulation dot1q
 switchport mode trunk
 no shutdown

! Configure Access Ports
interface range GigabitEthernet1/0 - 3
 switchport mode access
 switchport access vlan 10
 spanning-tree portfast
 no shutdown

! 802.1x Authentication (Optional based on server)
aaa new-model
aaa authentication dot1x default group radius
dot1x system-auth-control
radius-server host 192.168.20.100 key testing123

interface GigabitEthernet1/0
 dot1x pae authenticator
 dot1x port-control auto

end
write memory
